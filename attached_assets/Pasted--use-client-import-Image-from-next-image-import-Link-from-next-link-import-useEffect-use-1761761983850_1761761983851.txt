"use client";

import Image from "next/image";
import Link from "next/link";
import { useEffect, useMemo, useState } from "react";
import {
  Carousel,
  CarouselApi,
  CarouselContent,
  CarouselItem,
} from "@/components/ui/carousel";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";

// If you prefer, you can fetch this from an API or load it with dynamic import.
import servicesData from "@/data/services.json";

type Service = {
  id: string;
  title: string;
  summary?: string;
  badge?: string;
  href?: string;
  image?: string; // local path or remote URL
};

function Case() {
  const [api, setApi] = useState<CarouselApi>();
  const [current, setCurrent] = useState(0);

  // Basic safety guard in case JSON is empty
  const services: Service[] = useMemo(
    () => (Array.isArray(servicesData?.services) ? servicesData.services : []),
    []
  );

  useEffect(() => {
    if (!api || services.length === 0) return;

    const timer = setTimeout(() => {
      const lastIndex = api.scrollSnapList().length - 1;
      const isLast = api.selectedScrollSnap() === lastIndex;

      if (isLast) {
        setCurrent(0);
        api.scrollTo(0);
      } else {
        api.scrollNext();
        setCurrent((c) => c + 1);
      }
    }, 1800); // a touch slower feels nicer for reading

    return () => clearTimeout(timer);
  }, [api, current, services.length]);

  return (
    <div className="w-full py-20 lg:py-40">
      <div className="container mx-auto">
        <div className="flex flex-col gap-10">
          <h2 className="text-xl md:text-3xl lg:text-5xl tracking-tighter lg:max-w-3xl font-regular text-left">
            Proven training programs for real behavior change
          </h2>

          <Carousel setApi={setApi} className="w-full">
            <CarouselContent>
              {services.map((svc) => (
                <CarouselItem
                  key={svc.id}
                  className="basis-3/4 sm:basis-1/2 md:basis-1/3 lg:basis-1/4 xl:basis-1/6"
                >
                  <Card className="h-full overflow-hidden">
                    {svc.image ? (
                      <div className="relative aspect-square">
                        <Image
                          src={svc.image}
                          alt={svc.title}
                          fill
                          className="object-cover"
                          sizes="(max-width: 768px) 50vw, (max-width: 1200px) 25vw, 16vw"
                        />
                      </div>
                    ) : (
                      <div className="aspect-square bg-muted" />
                    )}
                    <CardHeader className="p-4">
                      <CardTitle className="text-base font-semibold tracking-tight">
                        {svc.title}
                      </CardTitle>
                      {svc.badge ? (
                        <span className="mt-1 inline-flex w-fit rounded-full bg-accent px-2 py-0.5 text-xs text-accent-foreground">
                          {svc.badge}
                        </span>
                      ) : null}
                    </CardHeader>
                    <CardContent className="p-4 pt-0">
                      {svc.summary ? (
                        <p className="text-sm text-muted-foreground line-clamp-3">
                          {svc.summary}
                        </p>
                      ) : null}

                      {svc.href ? (
                        <Link
                          href={svc.href}
                          className="mt-3 inline-flex text-sm underline underline-offset-4"
                        >
                          See program
                        </Link>
                      ) : null}
                    </CardContent>
                  </Card>
                </CarouselItem>
              ))}
            </CarouselContent>
          </Carousel>
        </div>
      </div>
    </div>
  );
}

export { Case };
